! Title: Instagram Reels Blocker
! Description: Hide reels and suggested content
instagram.com##a[href^="/reels"]
instagram.com##article:has(a[href*="/reel/"])
! Mobile web only. main > nav holds the search box on /explore/ and no other page.
! Not path-scoped on purpose: Instagram navigates client-side, so :matches-path would not re-apply.
instagram.com##main[role="main"]:has(> nav input) > div:has(a[href^="/p/"])
! "Use the app" bar above the bottom navigation, on every page that shows it.
instagram.com##section:has(> main) > div > div[data-visualcompletion="ignore"]
! Home feed: suggested posts are the ones with a Follow button in their header.
! Not keyed on the "caught up" marker: the feed unmounts it on scroll and the posts come back.
instagram.com##main[role="main"] article:has([data-interactable] > div:nth-child(2) [role="button"])
! "Suggested Posts" heading under the "You're all caught up" message.
instagram.com##main[role="main"] div:has(> div > div > div > img[src*="illo-confirm-refresh"]) > div:nth-child(2)
